def luhn_digit(body):
    """Контрольная цифра по алгоритму Луна для числа без неё."""
    total = 0
    for i, ch in enumerate(reversed(body)):
        d = int(ch)
        if i % 2 == 0:
            d *= 2
            if d > 9:
                d -= 9
        total += d
    return str((10 - total % 10) % 10)


def make_pan(system, serial):
    """Номер карты: БИН платёжной системы + 9 цифр порядкового номера + контрольная цифра."""
    body = BINS[system] + str(serial).zfill(9)
    return body + luhn_digit(body)


def make_account_number(bic, serial):
    """Номер счёта: 40817 810 К 0000 и 7 цифр порядкового номера, К считается по правилам ЦБ."""
    head = "40817810"
    tail = "0000" + str(serial).zfill(7)
    probe = bic[-3:] + head + "0" + tail
    weights = [7, 1, 3] * 8
    total = sum(int(d) * w % 10 for d, w in zip(probe, weights))
    return head + str(total % 10 * 3 % 10) + tail


_clock = [datetime(2025, 1, 10, 9, 30)]


def tick():
    """Следующий момент времени: каждый вызов позже предыдущего."""
    _clock[0] += timedelta(hours=7, minutes=13, seconds=22)
    return _clock[0]


def issue_and_expiry():
    """Дата выпуска карты (берётся из часов) и дата окончания срока: через 4 года."""
    issued = tick().date()
    return issued, issued.replace(year=issued.year + 4)

    letters = "абвгдеёжзийклмнопрстуфхцчшщъыьэюя-"

    def _check_amount(self, amount):
        if not isinstance(amount, (int, float)) or amount <= 0:
            raise ValidationError(ValidationError.AMOUNT_INVALID)

    def get_balance(self):
        self._ensure_usable()
        return f"Баланс: {self.account.balance:.2f}₽"

    def deposit(self, amount):
        self._check_amount(amount)
        self._ensure_usable()
        if amount > DEPOSIT_LIMIT:
            self.status = CardStatus.BLOCKED
            raise BusinessRuleError(BusinessRuleError.DEPOSIT_LIMIT)
        self.account.balance += amount
        self._log(TransactionType.DEPOSIT, amount, to_card=self.card_id,
                  description=f"Пополнение {amount:.2f}₽ на карту #{self.card_id}")

    def _check_pay(self, amount, mcc):
        self._check_amount(amount)
        if not (isinstance(mcc, str) and len(mcc) == 4 and mcc.isdigit()):
            raise ValidationError(ValidationError.MCC_INVALID)
        self._ensure_usable()
        if mcc in FORBIDDEN_MCC:
            raise BusinessRuleError(BusinessRuleError.MCC_FORBIDDEN.format(mcc=mcc))
        if amount > PAY_LIMIT:
            self.status = CardStatus.BLOCKED
            raise BusinessRuleError(BusinessRuleError.PAY_LIMIT)
        if self.account.balance < amount:
            raise InsufficientFundsError(InsufficientFundsError.FOR_PAY)

    def pay(self, amount, mcc):
        self._check_pay(amount, mcc)
        self.account.balance -= amount
        self._log(TransactionType.PAY, amount, from_card=self.card_id, mcc=mcc,
                  description=f"Покупка {amount:.2f}₽, MCC {mcc}")

    def transfer(self, to_card, amount):
        self._check_amount(amount)
        if not isinstance(to_card, Card):
            raise NotFoundError(NotFoundError.CARD_NOT_FOUND)
        self._ensure_usable()
        if to_card.account is None:
            raise AccessError(AccessError.RECIPIENT_NO_ACCOUNT)
        if to_card.status != CardStatus.ACTIVE:
            raise AccessError(AccessError.RECIPIENT_INACTIVE)
        if to_card is self:
            raise BusinessRuleError(BusinessRuleError.SELF_TRANSFER)
        if amount > TRANSFER_LIMIT:
            self.status = CardStatus.BLOCKED
            raise BusinessRuleError(BusinessRuleError.TRANSFER_LIMIT)
        if self.account.balance < amount:
            raise InsufficientFundsError(InsufficientFundsError.FOR_TRANSFER)
        self.account.balance -= amount
        to_card.account.balance += amount
        self._log(TransactionType.TRANSFER, amount, from_card=self.card_id, to_card=to_card.card_id,
                  description=f"Перевод {amount:.2f}₽: карта #{self.card_id} → карта #{to_card.card_id}")

    def apply_for_card(self, last_name, first_name, pin, phone, payment_system=DEFAULT_SYSTEM,
                       card_class=Card, **kwargs):
        user = None
        for u in self.customers.values():
            if (u.last_name, u.first_name, u.phone) == (last_name, first_name, phone):
                user = u
        is_new = user is None
        if is_new:
            user = User(len(self.customers) + 1, last_name, first_name, pin, phone)
        account = Account(make_account_number(self.bic, len(self.accounts) + 1), user)
        card_id = len(self.cards) + 1
        issued, expiry = issue_and_expiry()
        card = card_class(card_id=card_id, account=account, bank=self, payment_system=payment_system,
                          pan=make_pan(payment_system, card_id), issue_date=issued,
                          expiry_date=expiry, **kwargs)
        if is_new:
            self.customers[user.user_id] = user
        self.accounts[account.acc_id] = account
        self.cards[card_id] = card
        user.accounts.append(account)
        user.cards.append(card)
        return card

    def apply_for_card(self, last_name, first_name, pin, phone, payment_system=DEFAULT_SYSTEM,
                       card_class=Card, **kwargs):
        if not (isinstance(pin, str) and len(pin) == 4):
            raise ValidationError(ValidationError.PIN_INVALID)
        if not pin.isdigit():
            raise ValidationError(ValidationError.PIN_FORMAT)
        if not (is_russian_name(last_name) and is_russian_name(first_name)):
            raise ValidationError(ValidationError.NAME_INVALID)
        if payment_system not in BINS:
            raise ValidationError(ValidationError.SYSTEM_UNKNOWN.format(system=payment_system))
        user = None
        for u in self.customers.values():
            if u.phone == phone:
                if (u.last_name, u.first_name) != (last_name, first_name):
                    raise BusinessRuleError(BusinessRuleError.PHONE_TAKEN)
                user = u
        if user is not None and len(user.cards) >= MAX_CARDS:
            raise BusinessRuleError(BusinessRuleError.TOO_MANY_CARDS.format(n=MAX_CARDS))
        is_new = user is None
        if is_new:
            user = User(len(self.customers) + 1, last_name, first_name, pin, phone)
        account = Account(make_account_number(self.bic, len(self.accounts) + 1), user)
        card_id = len(self.cards) + 1
        issued, expiry = issue_and_expiry()
        card = card_class(card_id=card_id, account=account, bank=self, payment_system=payment_system,
                          pan=make_pan(payment_system, card_id), issue_date=issued,
                          expiry_date=expiry, **kwargs)
        if is_new:
            self.customers[user.user_id] = user
        self.accounts[account.acc_id] = account
        self.cards[card_id] = card
        user.accounts.append(account)
        user.cards.append(card)
        return card

import sys
from collections import Counter
from functools import wraps

errors = Counter()
KINDS = {"simple": SimpleDebitCard, "cashback": CashbackCard, "savings": SavingsCard}


def guard(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except BankError as err:
            errors[type(err).__name__] += 1
            return f"Ошибка ({type(err).__name__}): {err}"
    return wrapper


def number(text):
    try:
        return float(text)
    except ValueError:
        raise ValidationError(ValidationError.NOT_A_NUMBER)


def find_card(bank, text):
    try:
        card_id = int(text)
    except ValueError:
        raise ValidationError(ValidationError.NOT_A_NUMBER)
    if card_id not in bank.cards:
        raise NotFoundError(NotFoundError.NO_SUCH_CARD)
    return bank.cards[card_id]


@guard
def execute(bank, line):
    parts = line.split()
    if not parts:
        raise ValidationError(ValidationError.BAD_COMMAND)
    cmd, args = parts[0], parts[1:]
    if cmd == "apply" and len(args) in (5, 6):
        kind = args[5] if len(args) == 6 else "simple"
        if kind not in KINDS:
            raise ValidationError(ValidationError.BAD_COMMAND)
        card = bank.apply_for_card(*args[:5], card_class=KINDS[kind])
        return f"Выпущена карта #{card.card_id} ({kind}), PAN {card.pan}"
    if cmd == "deposit" and len(args) == 2:
        card = find_card(bank, args[0])
        card.deposit(number(args[1]))
        return f"Готово. Карта #{card.card_id}: {card.get_balance()}"
    if cmd == "pay" and len(args) == 3:
        card = find_card(bank, args[0])
        card.pay(number(args[1]), args[2])
        return f"Готово. Карта #{card.card_id}: {card.get_balance()}"
    if cmd == "transfer" and len(args) == 3:
        card = find_card(bank, args[0])
        card.transfer(find_card(bank, args[1]), number(args[2]))
        return f"Готово. Карта #{card.card_id}: {card.get_balance()}"
    if cmd == "balance" and len(args) == 1:
        card = find_card(bank, args[0])
        return f"Карта #{card.card_id}: {card.get_balance()}"
    if cmd == "info" and len(args) == 1:
        return find_card(bank, args[0]).get_card_info()
    if cmd == "history" and len(args) == 1:
        card = find_card(bank, args[0])
        return "\n".join([f"История карты #{card.card_id}:"] + (card.get_history() or ["(операций нет)"]))
    if cmd == "close" and len(args) == 1:
        card = find_card(bank, args[0])
        card.close()
        return f"Карта #{card.card_id} закрыта"
    if cmd == "accrue" and len(args) == 1:
        card = find_card(bank, args[0])
        if not isinstance(card, SavingsCard):
            raise BusinessRuleError(BusinessRuleError.NOT_SAVINGS)
        card.accrue_interest()
